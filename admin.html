addBtn.onclick=async()=>{ addMsg.textContent=''; const productName=document.getElementById('name').value.trim(); const productDescription=document.getElementById('description').value.trim(); const productPrice=document.getElementById('price').value.trim(); const file=document.getElementById('image').files[0]; if(!productName){ addMsg.textContent='Enter a product name.'; return; } if(!file){ addMsg.textContent='Choose a product image.'; return; } const ext=(file.name.split('.').pop()||'jpg').toLowerCase(); const path=crypto.randomUUID()+'.'+ext; const up=await db.storage .from('product-images') .upload(path,file,{upsert:false}); if(up.error){ addMsg.textContent=up.error.message; return; } const pub=db.storage .from('product-images') .getPublicUrl(path).data.publicUrl; const ins=await db.from('products').insert({ name:productName, description:productDescription, price:productPrice, image_url:pub, storage_path:path }); if(ins.error){ await db.storage.from('product-images').remove([path]); addMsg.textContent=ins.error.message; return; } document.getElementById('name').value=''; document.getElementById('description').value=''; document.getElementById('price').value=''; document.getElementById('image').value=''; addMsg.textContent='Product added successfully.'; load(); };
